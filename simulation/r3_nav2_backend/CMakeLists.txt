cmake_minimum_required(VERSION 3.16)
project(tai_r3_nav2_backend)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(nav2_util REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(pluginlib REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
add_executable(r3_nav2_backend main.cpp)
ament_target_dependencies(r3_nav2_backend rclcpp rclcpp_lifecycle nav2_util nav2_core nav2_costmap_2d nav2_msgs pluginlib tf2_ros tf2_geometry_msgs)
